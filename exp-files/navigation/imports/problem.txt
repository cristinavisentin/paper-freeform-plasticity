$testEnviroment = (arena = $testArenas)
* [ds.e.navigation(randomGenerator = m.sharedRG())]

$qFunctions = f.allNamed(
  fs = m.sKeyMap(
    keys = [maturePerformance; increase; inconsistency];
    values = [
      f.avg(of = f.subList(from = 0.9; to = 1; relative = true));
      ea.f.numExpr(
        expr = "-(x2;x1)";
        args = [
          f.avg(of = f.subList(from = 0; to = 0.1; relative = true));
          f.avg(of = f.subList(from = 0.9; to = 1; relative = true))
        ]
      );
      f.sd(of = f.subList(from = 0.9; to = 1; relative = true))
    ]
  );
  of = ds.f.rewards()
)

$problem = ea.p.simsToTo(
  name = "nav-lives";
  simulations = 3 * [ds.sat.sequential(
    tasks = 50 * [ds.srlat.fromNumericalEnvironment(
      environment = ds.e.navigation(arena = $a_barrier);
      reward = ds.e.nav.reward.reaching()
    )];
    resetFirst = true
  )];
  dT = $dT;
  tRange = $tRange;
  comparator = ea.comparator.descending(of = f.mapValue(key = maturePerformance));
  qFunction = $qFunctions;
  aggregator = f.mapAggregate(aggregateF = f.avg())
)
